// Advance Signal Calculation Logic function calculateAdvanceSignal(rsi, currentPrice, lastPrice) { let strength = 0; // 1. RSI Logic if (rsi < 30) strength += 40; // Oversold (Buy) if (rsi > 70) strength -= 40; // Overbought (Sell) // 2. Trend Logic (Price Action) if (currentPrice > lastPrice) strength += 20; // Upward trend else strength -= 20; // Downward trend // 3. Output if (strength >= 40) return "↑ STRONG CALL"; if (strength <= -40) return "↓ STRONG PUT"; return "⌛ WAIT - NO SIGNAL"; }
